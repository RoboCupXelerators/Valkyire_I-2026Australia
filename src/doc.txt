	./scripts/install_system_service.sh

查看攻防输出
	journalctl -u ENDFIELD -f

取消main.py攻防自动运行
	systemctl disable ENDFIELD.service

  # 机体坐标系 (不补偿航向)

  # 场地坐标系 (自动补偿航向)
  chassis.FieldMoveVector(SpeedX, SpeedY, FacingAngle, Kp=None, KpW=None)
  # FacingAngle: 车头朝向的绝对角度

  chassis.FieldMoveAngle(FacingAngle, MovingAngle, Speed, Kp=None)
  # 面朝FacingAngle, 往MovingAngle方向移动

  chassis.FieldTurn(Angle)
  # 原地转向到绝对角度 (有闭环)

  chassis.TurnTo(Angle, Kp=None)
  # 阻塞式转向 (慎用)

  chassis.stop()
  # 立即停车
